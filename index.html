import network
import time

def connect_wifi(ssid, password, timeout=10):
    """SSID와 비밀번호로 Wi-Fi에 접속하고, 성공 시 IP를 반환합니다."""
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)

    if wlan.isconnected():
        return wlan.ifconfig()[0]

    wlan.connect(ssid, password)
    start = time.time()
    while not wlan.isconnected():
        if time.time() - start > timeout:
            raise RuntimeError("Wi-Fi 접속 실패")
        time.sleep(0.5)

    return wlan.ifconfig()[0]


# 사용 예
ip = connect_wifi("여기에_SSID_입력", "여기에_비밀번호_입력")
print("IP:", ip)

Unit 1-6. CSV로 데이터 기록하기

🤖샘플 프롬프트 1-6-A
Raspberry Pi Pico 2 W에서 MicroPython으로 동작하는 코드를 만들어 주세요. 별도 모듈에 정의된 `from wifi_helper import connect_wifi` 함수를 그대로 사용해 Wi-Fi에 접속한 뒤, **1초마다 현재 연결된 Wi-Fi의 RSSI를 측정**하고, 보드 루트의 `rssi.csv` 파일에 한 줄씩 **누적**(append) 기록해 주세요. 파일이 처음 만들어질 때만 `t_ms,ssid,rssi` 헤더를 한 줄 적습니다. `time.ticks_ms()`로 시작 시점을 0으로 잡고 경과 시간을 밀리초로 기록합니다. SSID는 `wlan.config('essid')`로 가져옵니다. **60회** 측정한 뒤 파일을 닫고 종료합니다. 외부 라이브러리는 `network`, `time`만 사용하고, 중간에 정지 버튼이 눌려도 파일이 닫히도록 `try / finally`로 감싸 주세요. 코드 블록 하나로, 한국어 주석을 한 줄씩 달아 주세요.
코드 1-20PYTHON
from wifi_helper import connect_wifi
import network, time, os

SSID = "여기에_SSID_입력"
PASSWORD = "여기에_비밀번호_입력"

FILE = "rssi.csv"
N = 60                                      # 측정 횟수

ip = connect_wifi(SSID, PASSWORD)           # Wi-Fi 접속
wlan = network.WLAN(network.STA_IF)
essid = wlan.config('essid')                # 접속된 AP 이름

# 파일이 처음이면 헤더 한 줄, 아니면 append로 이어쓰기
need_header = FILE not in os.listdir()
f = open(FILE, "a")
try:
    if need_header:
        f.write("t_ms,ssid,rssi\n")
    start = time.ticks_ms()
    for i in range(N):
        t = time.ticks_diff(time.ticks_ms(), start)
        rssi = wlan.status('rssi')
        line = f"{t},{essid},{rssi}\n"
        f.write(line)
        f.flush()                           # 매 줄마다 디스크 쓰기 보장
        print(line.strip())                 # 셸에서 진행 상황 확인
        time.sleep(1)
finally:
    f.close()                               # 정지 버튼이 눌려도 닫힘 보장
    print(f"# 저장 완료: {FILE}")

from wifi_helper import connect_wifi
import network, socket

SSID = "여기에_SSID_입력"
PASSWORD = "여기에_비밀번호_입력"

ip = connect_wifi(SSID, PASSWORD)            # Wi-Fi 접속

HTML = "<html><body><h1>Hello, Pico!</h1></body></html>"

# 80번 포트로 들어오는 연결을 기다릴 소켓 준비
addr = socket.getaddrinfo("0.0.0.0", 80)[0][-1]
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind(addr)                            # 주소 점유
server.listen(1)                             # 한 명씩 받기

try:
    while True:
        client, who = server.accept()        # 누가 접속할 때까지 대기
        try:
            request = client.recv(1024)      # 요청 1KB 받기 (본문은 안 씀)

                "HTTP/1.1 200 OK\r\n"
                "Content-Type: text/html\r\n"
                "Connection: close\r\n\r\n"

            client.close()                   # 그 클라이언트와의 통신 종료

🤖샘플 프롬프트 1-8-B
이전 단계의 Pico 2 W MicroPython 웹 서버를 확장해 주세요. 매 요청마다 `wlan = network.WLAN(network.STA_IF)`의 **`wlan.scan()`**으로 주변 AP 목록을 받아, **RSSI가 강한 상위 5개**를 골라 다음과 같은 표 형식의 HTML로 응답합니다. ``` SSID RSSI 막대 MyWiFi -58 dBm ████████████████████ NeighborAP -71 dBm █████████████ ``` 막대는 `█` 블록 문자로 만들고, **-90 dBm을 0칸**, **-30 dBm을 30칸**으로 선형 환산합니다. 한 칸당 2 dBm 간격입니다. from wifi_helper import connect_wifi
import network, socket

SSID = "여기에_SSID_입력"
PASSWORD = "여기에_비밀번호_입력"

ip = connect_wifi(SSID, PASSWORD)
print("대시보드 주소: http://" + ip)

wlan = network.WLAN(network.STA_IF)

def bar(rssi, width=30, lo=-90, hi=-30):
    """-90 dBm = 0칸, -30 dBm = width칸으로 선형 환산"""
    if rssi <= lo: return ""
    if rssi >= hi: return "█" * width
    n = int((rssi - lo) / (hi - lo) * width)
    return "█" * n

def make_page():
    rows = []
    for ap in wlan.scan():                                    # (ssid, bssid, channel, rssi, ...)
        ssid = ap[0].decode() if ap[0] else "<hidden>"
        rssi = ap[3]
        rows.append((ssid, rssi))
    rows.sort(key=lambda r: r[1], reverse=True)               # RSSI 큰(=강한) 순
    top = rows[:5]

    lines = [f"{'SSID':<16}{'RSSI':<10}막대"]
    for ssid, rssi in top:
        lines.append(f"{ssid[:16]:<16}{rssi:>4} dBm   {bar(rssi)}")
    body = "<pre>" + "\n".join(lines) + "</pre>"
    return "<html><body>" + body + "</body></html>"

addr = socket.getaddrinfo("0.0.0.0", 80)[0][-1]
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind(addr); server.listen(1)

try:
    while True:
        client, _ = server.accept()
        try:
            client.recv(1024)
            body = make_page()
            resp = ("HTTP/1.1 200 OK\r\n"
                    "Content-Type: text/html\r\n"

from wifi_helper import connect_wifi
import network, socket

SSID = "여기에_SSID_입력"
PASSWORD = "여기에_비밀번호_입력"
REFRESH = 2                                                   # 새로 고침 간격(초)

ip = connect_wifi(SSID, PASSWORD)
print("대시보드 주소: http://" + ip)

wlan = network.WLAN(network.STA_IF)

def bar(rssi, width=30, lo=-90, hi=-30):
    if rssi <= lo: return ""
    if rssi >= hi: return "█" * width
    return "█" * int((rssi - lo) / (hi - lo) * width)

def make_page():
    rows = []
    for ap in wlan.scan():
        ssid = ap[0].decode() if ap[0] else "<hidden>"
        rows.append((ssid, ap[3]))
    rows.sort(key=lambda r: r[1], reverse=True)
    top = rows[:5]

    lines = [f"{'SSID':<16}{'RSSI':<10}막대"]
    for ssid, rssi in top:
        lines.append(f"{ssid[:16]:<16}{rssi:>4} dBm   {bar(rssi)}")

    head = (f'<head><meta http-equiv="refresh" content="{REFRESH}">'
            f'<title>Pico Wi-Fi Dashboard</title></head>')
    body = "<body><h2>Wi-Fi 감도 대시보드</h2><pre>" + "\n".join(lines) + "</pre></body>"
    return "<html>" + head + body + "</html>"

addr = socket.getaddrinfo("0.0.0.0", 80)[0][-1]
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.bind(addr); server.listen(1)

try:
    while True:
        client, _ = server.accept()
        try:
            client.recv(1024)
            body = make_page()
            resp = ("HTTP/1.1 200 OK\r\n"
                    "Content-Type: text/html\r\n"
                    "Connection: close\r\n\r\n" + body)
            client.send(resp.encode())
        finally:
            client.close()
except KeyboardInterrupt:
    print("\n서버 종료")
finally:
    server.close()

3.4 둘을 잇기 · LED 게이지

코드 7 · 가스 값 → LED 게이지 (main.py)PYTHON
from machine import ADC, Pin
from neopixel import NeoPixel
import time
gas = ADC(Pin(26))
np = NeoPixel(Pin(16), 10)
SAFE = 15000      # 이 아래는 안전
DANGER = 30000    # 이 위는 위험
def read_average(s, count=10):
    total = 0
    for _ in range(count):
        total += s.read_u16(); time.sleep_ms(10)
    return total // count
def show_gauge(value):
    # 값을 0~10칸으로 바꾸기
    level = int((value - SAFE) / (DANGER - SAFE) * 10)
    level = max(0, min(10, level))
    for i in range(10):
        if i >= level:        np[i] = (0, 0, 0)
        elif i < 6:           np[i] = (0, 40, 0)   # 초록
        elif i < 8:           np[i] = (40, 30, 0)  # 노랑
        else:                 np[i] = (40, 0, 0)   # 빨강
    np.write()
while True:
    value = read_average(gas, 10)
    print(value)
    if value >= DANGER:
        for _ in range(3):                 # 위험! 전체 빨강 깜빡
            for i in range(10): np[i] = (60, 0, 0)
            np.write(); time.sleep(0.2)
            for i in range(10): np[i] = (0, 0, 0)
            np.write(); time.sleep(0.2)
    else:
        show_gauge(value)
    time.sleep(0.5)
