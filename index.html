CHAPTER 01

피코, 처음 시작하기

피코·MicroPython·Thonny 설치와 그로브 쉴드 이해, 그리고 보드 위 LED 깜빡이기 — 가장 처음 만나는 코드입니다.

첫 코드 · print 와 보드 LED 깜빡이기

코드 1-01PYTHON
>>> print("안녕, 피코!")
코드 1-02PYTHON
from machine import Pin   # 핀(Pin) 제어 도구를 가져옵니다.
import time               # 시간(time) 제어 도구를 가져옵니다.

led = Pin("LED", Pin.OUT) # 온보드 LED를 출력(OUT) 모드로 설정합니다.

led.value(1)              # LED를 켭니다. (1: 켜짐)
time.sleep(1)             # 1초 동안 대기합니다.
led.value(0)              # LED를 끕니다. (0: 꺼짐)
코드 1-03PYTHON
from machine import Pin
import time

led = Pin("LED", Pin.OUT)

while True:               # 아래 줄들을 계속 반복합니다.
    led.value(1)          # LED 켜기
    time.sleep(0.5)       # 0.5초 대기
    led.value(0)          # LED 끄기
    time.sleep(0.5)       # 0.5초 대기
코드 1-04PYTHON
from machine import Pin
import time

led = Pin("LED", Pin.OUT)
delay = 0.5               # 깜빡임 간격(초)을 변수로 관리

while True:
    led.value(1)
    time.sleep(delay)
    led.value(0)
    time.sleep(delay)

import network, time

6단계 · 살아 움직이는 대시보드로 업그레이드

🤖AI 도구에게 이렇게 부탁해 보세요
지금 만든 피코 웹앱을 업그레이드해 줘. 1) 새로고침을 누르지 않아도 5초마다 화면이 저절로 갱신되게 해 줘. 2) 최근 20개의 신호 세기 값을 막대그래프로 보여 줘. 3) 다만 피코가 버거워하지 않도록, 주변 와이파이를 전부 훑는 작업은 30초에 한 번만 하게 해 줘.
코드 블록 3 · 피코 웹서버 (main.py)PYTHON
import network, socket, time
from wifi_config import SSID, PASSWORD
wlan = network.WLAN(network.STA_IF)
wlan.active(True)
wlan.connect(SSID, PASSWORD)
while not wlan.isconnected():
    time.sleep(1)
ip = wlan.ifconfig()[0]
print("브라우저에서 접속:", "http://" + ip)
history = []
def make_page(rssi, hist):
    bars = ""
    for v in hist:
        h = max(8, min(100, (v + 90) * 2))
        bars += "<div class='bar' style='height:%dpx'></div>" % h
    return PAGE % (rssi, bars)
s = socket.socket()
s.bind(("0.0.0.0", 80))
s.listen(1)
while True:
    cl, addr = s.accept()
    cl.recv(1024)
    rssi = wlan.status("rssi")
    history.append(rssi)
    if len(history) > 20:
        history.pop(0)
    cl.send("HTTP/1.1 200 OK\r\nContent-Type: text/html\r\n\r\n")
    cl.send(make_page(rssi, history))
    cl.close()

코드 2 · 전체를 한 색으로 (fill 함수)PYTHON
from machine import Pin
from neopixel import NeoPixel
import time
NUM = 10
np = NeoPixel(Pin(16), NUM)
def fill(color):
    for i in range(NUM):
        np[i] = color
    np.write()
fill((0, 40, 0))   # 전체 초록
time.sleep(1)
fill((0, 0, 0))    # 전체 끄기

def gauge(level):
    # level: 0~10, 켤 칸 수

            np[i] = (0, 0, 0)    # 꺼진 칸

for n in range(0, 11):

    total = 0

        total += sensor.read_u16()
        time.sleep_ms(10)

3.4 둘을 잇기 · LED 게이지

코드 7 · 가스 값 → LED 게이지 (main.py)PYTHON
from machine import ADC, Pin
from neopixel import NeoPixel
import time
gas = ADC(Pin(26))
np = NeoPixel(Pin(16), 10)
SAFE = 15000      # 이 아래는 안전
DANGER = 30000    # 이 위는 위험
def read_average(s, count=10):
    total = 0
    for _ in range(count):
        total += s.read_u16(); time.sleep_ms(10)
    return total // count
def show_gauge(value):
    # 값을 0~10칸으로 바꾸기
    level = int((value - SAFE) / (DANGER - SAFE) * 10)
    level = max(0, min(10, level))
    for i in range(10):
        if i >= level:        np[i] = (0, 0, 0)
        elif i < 6:           np[i] = (0, 40, 0)   # 초록
        elif i < 8:           np[i] = (40, 30, 0)  # 노랑
        else:                 np[i] = (40, 0, 0)   # 빨강
    np.write()
while True:
    value = read_average(gas, 10)
    print(value)
    if value >= DANGER:
        for _ in range(3):                 # 위험! 전체 빨강 깜빡
            for i in range(10): np[i] = (60, 0, 0)
            np.write(); time.sleep(0.2)
            for i in range(10): np[i] = (0, 0, 0)
            np.write(); time.sleep(0.2)
    else:
        show_gauge(value)
    time.sleep(0.5)
